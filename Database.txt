5. CREATE TABLE (สร้างตาราง)
CREATE TABLE tbl_users (
id INT AUTO_INCREMENT PRIMARY KEY,
firstname VARCHAR(100),
fullname VARCHAR(200),
lastname VARCHAR(100),
username VARCHAR(100),
password VARCHAR(255),
status VARCHAR(20),
created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP
); 
6. SHOW TABLES (แสดงตาราง)
show tables;

show columns from tbl_users;

7. INSERT (Create - เพิ่มข้อมูล)
INSERT INTO tbl_users (firstname, fullname, lastname, username, password, status) VALUES ('Anuchat', 'Anuchat Rungsiyanon', 'Rungsiyanon', 'anuchat', '123456', 'user');
💡 ถ้ามีการเข้ารหัสรหัสผ่าน (hash) เช่น SHA2():

 
INSERT INTO tbl_users (firstname, fullname, lastname, username, password, status) VALUES ('Anuchat', 'Anuchat Rungsiyanon', 'Rungsiyanon', 'anuchat', SHA2('123456', 256), 'user');
8. SELECT (Read - อ่านข้อมูล)
-- ดึงข้อมูลทั้งหมด
SELECT * FROM tbl_users;

-- ดึงข้อมูลเฉพาะ id ที่ต้องการ
SELECT * FROM tbl_users WHERE id = 1;

-- ดึงเฉพาะบางคอลัมน์
SELECT id, fullname, username, status FROM tbl_users;
9. UPDATE (Update - แก้ไขข้อมูล)
UPDATE tbl_users SET firstname = 'Anuchat', lastname = 'Rungsiyanon', fullname = 'Anuchat Rungsiyanon', username = 'anuchat', status = 'user' WHERE id = 1;
ถ้าต้องการอัปเดตรหัสผ่านใหม่ (แบบเข้ารหัส):

 
UPDATE tbl_users SET password = SHA2('newpassword', 256) WHERE username = 'anuchat';
10. DELETE (Delete - ลบข้อมูล)
DELETE FROM tbl_users WHERE id = 1;
ถ้าอยากลบทั้งหมด (ระวัง!)

 
DELETE FROM tbl_users; //ลบแบบนับ id ไปเรื่อยๆ
TRUNCATE TABLE tbl_users; //ลบแบบ reset id
11. CREATE TABLE (สร้างตาราง)
CREATE TABLE tbl_orders (
  id INT AUTO_INCREMENT PRIMARY KEY,
  user_id INT NOT NULL,
  order_number VARCHAR(50) UNIQUE,
  total_amount DECIMAL(10,2) DEFAULT 0.00,
  payment_status ENUM('pending','paid','cancelled','refunded') DEFAULT 'pending',
  order_status ENUM('processing','shipped','delivered','cancelled') DEFAULT 'processing',
  payment_method VARCHAR(50),
  shipping_address TEXT,
  note TEXT,
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP
);

#อธิบายฟิลด์แต่ละช่อง
ฟิลด์	ประเภทข้อมูล	รายละเอียด
id	INT, PK	รหัสคำสั่งซื้อ (Primary Key)
user_id	INT	เชื่อมโยงกับผู้ใช้ใน tbl_users
order_number	VARCHAR(50)	หมายเลขคำสั่งซื้อ เช่น ORD20251110-001
total_amount	DECIMAL(10,2)	ยอดรวมคำสั่งซื้อ
payment_status	ENUM	สถานะการชำระเงิน (pending, paid, cancelled, refunded)
order_status	ENUM	สถานะคำสั่งซื้อ (processing, shipped, delivered, cancelled)
payment_method	VARCHAR(50)	วิธีการชำระ เช่น credit_card, bank_transfer, cash
shipping_address	TEXT	ที่อยู่จัดส่ง
note	TEXT	หมายเหตุจากผู้ใช้
created_at	TIMESTAMP	วันที่สั่งซื้อ
updated_at	TIMESTAMP	วันที่อัปเดตล่าสุด

12. INER JOIN (เรียกข้อมูล 2 ตาราง)
SELECT u.fullname, u.lastname, o.total_amount FROM tbl_users AS u INNER JOIN tbl_orders AS o ON u.id = o.user_id
